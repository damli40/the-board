# The Board — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A two-party agreement surface where each side's agent can only call tools that a currently-active, versioned rule authorises — so a rule applied retroactively is structurally unmakeable.

**Architecture:** One tab, one frame tree. A parent origin owns the record, the rule set and the tool registry. Two cross-origin iframes hold one agent panel each, isolated so neither side's agent can see the other's tools. A rule's lifetime is an `AbortController`: superseding a rule aborts it, and every tool that rule authorised leaves the agents' tool maps. The UI renders as a board — a rule track, moves as pieces stamped with the version live when played, and each side's current capability as a hand of cards.

**Tech Stack:** Vite + React + TypeScript · Tailwind + shadcn/ui · `motion` (ex-framer-motion) for the card-deal animation · Vitest for unit tests · Netlify (three sites, custom headers, Functions for provider key proxies) · WebMCP via Chrome 149 origin trial / `chrome://flags/#enable-webmcp-testing`

**Spec:** `docs/superpowers/specs/2026-08-26-the-board-design.md`
**Storyboard (drives Task 8-9 acceptance):** `docs/STORYBOARD.md`
**Evidence (drives the written explanation):** `docs/evidence/real-world-cases.md`

## Global Constraints

- **Deadline: submit Sep 2 2026.** Hard close is Sep 3, 1:00pm PDT (9:00pm Lagos). Do not plan into the final day.
- **Public repo with an OSS licence detectable in the GitHub About section** — an explicit rule, not a nicety. MIT, committed in Task 1.
- **`document.modelContext.registerTool({...})` must be visibly present in the repo** — named in the official rules.
- **No secrets in client code.** The repo is public. Every provider key lives in a Netlify Function.
- **Naming rule, binding on every artefact** (video, repo, README, submission text): no organisation, amount, sector, event type or counterparty from the source dispute. Shape only. See spec §3.
- **Never claim "two people, two browsers."** `exposedTo` scopes origins, not users. The claim is "two parties' agents, two origins, one record, co-present session."
- **Never pitch this as the gate the browser lacks.** ChatGPT's browser already applies confirmation policies to consequential actions and one judge leads it.
- Tool names: ≤128 chars, `[A-Za-z0-9_.-]` only.
- Origin isolation required: serve `Origin-Agent-Cluster: ?1`; never `?0`. GitHub Pages cannot do this — Netlify only.
- `Permissions-Policy: tools=(self "https://client.theboard.app" "https://contractor.theboard.app")` on the parent; `allow="tools"` on both iframes.

```ts
// packages/record/src/rules/model.ts
export type Side = 'client' | 'contractor';

export type ToolGrant = {
  name: string; title: string; description: string; side: Side; readOnly: boolean;
};

export type Rule = {
  id: string; text: string; version: number;
  activeFrom: number; activeUntil: number | null; grants: ToolGrant[];
};

export type RuleSet = { version: number; rules: Rule[] };

export function activeAt(set: RuleSet, at: number): Rule[] {
  return set.rules.filter(r =>
    r.activeFrom <= at && (r.activeUntil === null || at < r.activeUntil)
  );
}

export function supersede(
  set: RuleSet,
  ruleId: string,
  next: Omit<Rule, 'version' | 'activeFrom' | 'activeUntil'>,
  at: number
): RuleSet {
  const current = set.rules.find(r => r.id === ruleId && r.activeUntil === null);
  if (!current) throw new Error(`no live rule ${ruleId}`);
  if (at <= current.activeFrom) {
    throw new Error('forward-only: a rule change cannot predate the rule it supersedes');
  }
  const version = set.version + 1;
  return {
    version,
    rules: [
      ...set.rules.map(r => (r === current ? { ...r, activeUntil: at } : r)),
      { ...next, version, activeFrom: at, activeUntil: null }
    ]
  };
}
```

```ts
// packages/record/src/rules/refusal.ts
import { activeAt, type RuleSet, type Side } from './model';

export type Move = {
  id: string; side: Side; toolName: string; at: number; ruleSetVersion: number;
};
export type Claim = { moveId: string; ruleId: string };
export type Verdict = {
  admissible: boolean; reason: string;
  ruleVersionAtMove: number; ruleVersionOfClaim: number;
};

export function adjudicate(set: RuleSet, moves: Move[], claim: Claim): Verdict {
  const move = moves.find(m => m.id === claim.moveId);
  if (!move) throw new Error(`unknown move ${claim.moveId}`);

  const liveThen = activeAt(set, move.at).find(r => r.id === claim.ruleId);
  const claimed = set.rules
    .filter(r => r.id === claim.ruleId)
    .sort((a, b) => b.version - a.version)[0];
  if (!claimed) throw new Error(`unknown rule ${claim.ruleId}`);

  if (liveThen && liveThen.version === claimed.version) {
    return {
      admissible: true,
      reason: `Rule ${claim.ruleId} v${claimed.version} was live when move ${move.id} was played.`,
      ruleVersionAtMove: liveThen.version,
      ruleVersionOfClaim: claimed.version
    };
  }

  const grants = claimed.grants.map(g => g.name).join(', ') || 'none';
  return {
    admissible: false,
    reason:
      `Rule ${claim.ruleId} v${claimed.version} took effect after move ${move.id}. ` +
      `At that moment the rule set was v${move.ruleSetVersion}, and the capability it requires ` +
      `(${grants}) did not exist in the ${move.side}'s tool map.`,
    ruleVersionAtMove: liveThen?.version ?? 0,
    ruleVersionOfClaim: claimed.version
  };
}
```

```ts
// packages/record/src/rules/registry.ts
import { activeAt, type RuleSet, type Side } from './model';
import type { ModelContextLike } from '../webmcp/register';

type Origins = Record<Side, string>;
type Executor = (toolName: string, input: unknown) => Promise<string>;

export class ToolRegistry {
  private aborts = new Map<string, AbortController>();  // ruleKey -> controller
  private names = new Set<string>();

  constructor(
    private ctx: ModelContextLike,
    private origins: Origins,
    private execute: Executor
  ) {}

  registered(): string[] { return [...this.names]; }

  async sync(set: RuleSet, at: number): Promise<void> {
    const live = activeAt(set, at);
    const wanted = new Set(live.map(r => `${r.id}@${r.version}`));

    for (const [key, ac] of this.aborts) {
      if (!wanted.has(key)) { ac.abort(); this.aborts.delete(key); }
    }

    this.names.clear();
    for (const rule of live) {
      const key = `${rule.id}@${rule.version}`;
      for (const g of rule.grants) this.names.add(g.name);
      if (this.aborts.has(key)) continue;

      const ac = new AbortController();
      this.aborts.set(key, ac);
      for (const g of rule.grants) {
        await this.ctx.registerTool({
          name: g.name,
          title: g.title,
          description: `${g.description} (authorised by rule ${rule.id} v${rule.version})`,
          inputSchema: { type: 'object', properties: {} },
          annotations: { readOnlyHint: g.readOnly, untrustedContentHint: true },
          execute: (input) => this.execute(g.name, input)
        }, { signal: ac.signal, exposedTo: [this.origins[g.side]] });
      }
    }
  }
}
```

```ts
// packages/panel/src/agent/sanitize.ts
const OPEN = '<untrusted-counterparty-text>';
const CLOSE = '</untrusted-counterparty-text>';

const INJECTION_MARKERS = [
  /ignore\s+(all\s+)?previous\s+instructions?/gi,
  /disregard\s+(all\s+)?(prior|previous)\s+/gi,
  /system\s*:\s*/gi,
  /you\s+are\s+now\s+/gi
];

export function sanitizeCounterpartyText(raw: string): string {
  let body = raw.split(OPEN).join('').split(CLOSE).join('');
  for (const re of INJECTION_MARKERS) body = body.replace(re, '[redacted-instruction] ');
  return `${OPEN}${body}${CLOSE}`;
}
```

```ts
// netlify/functions/model-proxy.ts
import type { Handler } from '@netlify/functions';

export const handler: Handler = async (event) => {
  if (event.httpMethod !== 'POST') return { statusCode: 405, body: 'POST only' };
  const key = process.env.MODEL_API_KEY;                 // set per Netlify site
  const base = process.env.MODEL_BASE_URL;               // per-provider
  if (!key || !base) return { statusCode: 500, body: 'proxy not configured' };

  const upstream = await fetch(base, {
    method: 'POST',
    headers: { 'content-type': 'application/json', authorization: `Bearer ${key}` },
    body: event.body ?? '{}'
  });
  return {
    statusCode: upstream.status,
    headers: { 'content-type': 'application/json' },
    body: await upstream.text()
  };
};
```
